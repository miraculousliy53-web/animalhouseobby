  <script>
    // --- QUESTION DATASET (20 Curriculum Questions) ---
    const QUESTIONS = [
      { text: "There is a cat on the bed.", animal: "cat", count: 1, prep: "on", obj: "bed" },
      { text: "There are two dogs under the table.", animal: "dog", count: 2, prep: "under", obj: "table" },
      { text: "There is a rabbit in the box.", animal: "rabbit", count: 1, prep: "in", obj: "box" },
      { text: "There are three cats on the sofa.", animal: "cat", count: 3, prep: "on", obj: "sofa" },
      { text: "There is a frog under the chair.", animal: "frog", count: 1, prep: "under", obj: "chair" },
      { text: "There are four birds on the table.", animal: "bird", count: 4, prep: "on", obj: "table" },
      { text: "There is a dog in the basket.", animal: "dog", count: 1, prep: "in", obj: "basket" },
      { text: "There are two rabbits in the box.", animal: "rabbit", count: 2, prep: "in", obj: "box" },
      { text: "There is a kitten under the bed.", animal: "kitten", count: 1, prep: "under", obj: "bed" },
      { text: "There are three frogs on the table.", animal: "frog", count: 3, prep: "on", obj: "table" },
      { text: "There is a bird in the basket.", animal: "bird", count: 1, prep: "in", obj: "basket" },
      { text: "There are five cats on the bed.", animal: "cat", count: 5, prep: "on", obj: "bed" },
      { text: "There is a rabbit under the table.", animal: "rabbit", count: 1, prep: "under", obj: "table" },
      { text: "There are two dogs in the box.", animal: "dog", count: 2, prep: "in", obj: "box" },
      { text: "There is a kitten on the chair.", animal: "kitten", count: 1, prep: "on", obj: "chair" },
      { text: "There are two rabbits on the bed.", animal: "rabbit", count: 2, prep: "on", obj: "bed" },
      { text: "There is a frog under the chair.", animal: "frog", count: 1, prep: "under", obj: "chair" },
      { text: "There are three kittens in the basket.", animal: "kitten", count: 3, prep: "in", obj: "basket" },
      { text: "There are five birds on the table.", animal: "bird", count: 5, prep: "on", obj: "table" },
      { text: "There is a dog under the bed.", animal: "dog", count: 1, prep: "under", obj: "bed" }
    ];

    // --- GAME STATE ---
    let qIndex = 0;
    let score = 0;
    let timer = 60;
    let timerInterval = null;
    let wordBankState = []; // Master word bank objects: [{id, word, selected: false}]
    let userSentence = [];  // Array of selected word objects in order

    // --- SOUND ENGINE (Web Audio API) ---
    const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    function playAudioTone(freq, type, duration) {
      if (audioCtx.state === 'suspended') audioCtx.resume();
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.type = type;
      osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
      gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.start();
      osc.stop(audioCtx.currentTime + duration);
    }
    function soundPop() { playAudioTone(600, 'sine', 0.1); }
    function soundCorrect() {
      playAudioTone(523.25, 'triangle', 0.15);
      setTimeout(() => playAudioTone(659.25, 'triangle', 0.15), 100);
      setTimeout(() => playAudioTone(783.99, 'triangle', 0.3), 200);
    }
    function soundError() {
      playAudioTone(200, 'sawtooth', 0.2);
    }

    // --- GAME INITIATION & QUESTION FLOW ---
    function startGame() {
      document.getElementById('game-modal').classList.add('hidden');
      qIndex = 0;
      score = 0;
      updateHUD();
      loadQuestion();
    }

    function updateHUD() {
      document.getElementById('hud-qnum').innerText = `${qIndex + 1} / ${QUESTIONS.length}`;
      document.getElementById('hud-score').innerText = score;
    }

    function startTimer() {
      clearInterval(timerInterval);
      timer = 60;
      document.getElementById('hud-timer').innerText = timer;
      timerInterval = setInterval(() => {
        timer--;
        document.getElementById('hud-timer').innerText = timer;
        if (timer <= 0) {
          clearInterval(timerInterval);
          soundError();
          alert("Time's up! Let's try the question again.");
          loadQuestion();
        }
      }, 1000);
    }

    function loadQuestion() {
      const q = QUESTIONS[qIndex];
      updateHUD();
      startTimer();
      render3DScene(q);

      // Tokenize target sentence & create deterministic word bank objects
      const rawWords = q.text.split(" ");
      const shuffled = [...rawWords].sort(() => Math.random() - 0.5);

      wordBankState = shuffled.map((word, idx) => ({
        id: idx,
        word: word,
        selected: false
      }));

      userSentence = [];
      renderWordTiles();
    }

    // --- SAFE WORD BANK & SENTENCE RENDERER ---
    function renderWordTiles() {
      const bankContainer = document.getElementById('word-bank');
      const answerContainer = document.getElementById('answer-box');
      
      bankContainer.innerHTML = '';
      answerContainer.innerHTML = '';

      // Render Selected Answer Tiles
      if (userSentence.length === 0) {
        answerContainer.innerHTML = `<span id="placeholder-text" class="text-slate-500 text-sm italic">Tap words below to build your sentence...</span>`;
      } else {
        userSentence.forEach((item, index) => {
          const tile = document.createElement('button');
          tile.className = "word-tile bg-indigo-600 hover:bg-indigo-500 active:bg-indigo-700 text-white font-bold px-3.5 py-2 rounded-xl text-sm md:text-base shadow-md border border-indigo-400 flex items-center gap-1";
          tile.innerText = item.word;
          tile.onclick = () => removeWordFromSentence(index);
          answerContainer.appendChild(tile);
        });
      }

      // Render Available Word Bank Tiles
      wordBankState.forEach((item) => {
        if (!item.selected) {
          const tile = document.createElement('button');
          tile.className = "word-tile bg-slate-800 hover:bg-slate-700 active:bg-slate-900 text-yellow-300 font-bold px-3.5 py-2 rounded-xl text-sm md:text-base border border-slate-600 shadow";
          tile.innerText = item.word;
          tile.onclick = () => selectWordTile(item.id);
          bankContainer.appendChild(tile);
        }
      });
    }

    function selectWordTile(id) {
      soundPop();
      const targetItem = wordBankState.find(item => item.id === id);
      if (targetItem && !targetItem.selected) {
        targetItem.selected = true;
        userSentence.push(targetItem);
        renderWordTiles();
      }
    }

    function removeWordFromSentence(index) {
      soundPop();
      const removedItem = userSentence.splice(index, 1)[0];
      if (removedItem) {
        const originalStateItem = wordBankState.find(item => item.id === removedItem.id);
        if (originalStateItem) originalStateItem.selected = false;
      }
      renderWordTiles();
    }

    function clearSentence() {
      soundPop();
      wordBankState.forEach(item => item.selected = false);
      userSentence = [];
      renderWordTiles();
    }

    function submitAnswer() {
      const constructedText = userSentence.map(item => item.word).join(" ");
      const targetText = QUESTIONS[qIndex].text;

      if (constructedText === targetText) {
        clearInterval(timerInterval);
        soundCorrect();
        score += 100;
        updateHUD();

        if (qIndex + 1 < QUESTIONS.length) {
          qIndex++;
          setTimeout(() => {
            alert("Great job! Correct answer!");
            loadQuestion();
          }, 200);
        } else {
          showFinalResults();
        }
      } else {
        soundError();
        alert("Not quite right. Check your word order and try again!");
      }
    }

    function showFinalResults() {
      clearInterval(timerInterval);
      document.getElementById('modal-title').innerText = "QUEST COMPLETED!";
      document.getElementById('modal-desc').innerHTML = `You completed all 20 questions!<br><br><span class="text-2xl font-black text-emerald-400">Final Score: ${score} / 2000</span>`;
      document.getElementById('modal-btn').innerText = "PLAY AGAIN";
      document.getElementById('game-modal').classList.remove('hidden');
    }

    // --- THREE.JS 3D SCENE & MODEL GENERATOR ---
    let scene, camera, renderer, objectGroup;

    function init3D() {
      const container = document.getElementById('canvas-container');
      scene = new THREE.Scene();
      scene.background = new THREE.Color(0x111827);

      camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 1000);
      camera.position.set(0, 5, 10);
      camera.lookAt(0, 1, 0);

      renderer = new THREE.WebGLRenderer({ antialias: true });
      renderer.setSize(window.innerWidth, window.innerHeight);
      renderer.shadowMap.enabled = true;
      container.appendChild(renderer.domElement);

      // Lighting
      const ambientLight = new THREE.AmbientLight(0xffffff, 0.7);
      scene.add(ambientLight);

      const dirLight = new THREE.DirectionalLight(0xffffff, 0.6);
      dirLight.position.set(5, 12, 8);
      dirLight.castShadow = true;
      scene.add(dirLight);

      // Base Room Environment
      const floorGeo = new THREE.PlaneGeometry(12, 12);
      const floorMat = new THREE.MeshStandardMaterial({ color: 0x374151, roughness: 0.8 });
      const floor = new THREE.Mesh(floorGeo, floorMat);
      floor.rotation.x = -Math.PI / 2;
      floor.receiveShadow = true;
      scene.add(floor);

      objectGroup = new THREE.Group();
      scene.add(objectGroup);

      window.addEventListener('resize', onWindowResize);
      animate();
    }

    function animate() {
      requestAnimationFrame(animate);
      if (objectGroup) objectGroup.rotation.y += 0.003; // Gentle rotation
      renderer.render(scene, camera);
    }

    function onWindowResize() {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
    }

    // --- PROCEDURAL 3D FURNITURE & ANIMALS ---
    function render3DScene(q) {
      while(objectGroup.children.length > 0) { 
        objectGroup.remove(objectGroup.children[0]); 
      }

      // 1. Build Base Furniture
      let furnitureMesh = new THREE.Group();
      if (q.obj === "bed") {
        const frame = new THREE.Mesh(new THREE.BoxGeometry(4, 0.6, 3), new THREE.MeshStandardMaterial({color: 0x8B4513}));
        const mattress = new THREE.Mesh(new THREE.BoxGeometry(3.8, 0.5, 2.8), new THREE.MeshStandardMaterial({color: 0xffffff}));
        mattress.position.y = 0.55;
        furnitureMesh.add(frame, mattress);
      } else if (q.obj === "table") {
        const top = new THREE.Mesh(new THREE.BoxGeometry(4, 0.3, 3), new THREE.MeshStandardMaterial({color: 0x10B981}));
        top.position.y = 1.5;
        furnitureMesh.add(top);
        [[-1.8, -1.3], [1.8, -1.3], [-1.8, 1.3], [1.8, 1.3]].forEach(p => {
          const leg = new THREE.Mesh(new THREE.BoxGeometry(0.2, 1.5, 0.2), new THREE.MeshStandardMaterial({color: 0x059669}));
          leg.position.set(p[0], 0.75, p[1]);
          furnitureMesh.add(leg);
        });
      } else if (q.obj === "sofa") {
        const base = new THREE.Mesh(new THREE.BoxGeometry(4, 0.8, 2), new THREE.MeshStandardMaterial({color: 0x3B82F6}));
        const back = new THREE.Mesh(new THREE.BoxGeometry(4, 1.5, 0.5), new THREE.MeshStandardMaterial({color: 0x1D4ED8}));
        back.position.set(0, 0.75, -0.75);
        furnitureMesh.add(base, back);
      } else if (q.obj === "chair") {
        const seat = new THREE.Mesh(new THREE.BoxGeometry(2, 0.2, 2), new THREE.MeshStandardMaterial({color: 0xF59E0B}));
        seat.position.y = 1.0;
        furnitureMesh.add(seat);
        [[-0.8, -0.8], [0.8, -0.8], [-0.8, 0.8], [0.8, 0.8]].forEach(p => {
          const leg = new THREE.Mesh(new THREE.BoxGeometry(0.15, 1.0, 0.15), new THREE.MeshStandardMaterial({color: 0xD97706}));
          leg.position.set(p[0], 0.5, p[1]);
          furnitureMesh.add(leg);
        });
      } else if (q.obj === "box" || q.obj === "basket") {
        const box = new THREE.Mesh(new THREE.BoxGeometry(3, 1.2, 3), new THREE.MeshStandardMaterial({color: 0xD97706, roughness: 0.9}));
        box.position.y = 0.6;
        furnitureMesh.add(box);
      }
      objectGroup.add(furnitureMesh);

      // 2. Build & Place Animals
      for (let i = 0; i < q.count; i++) {
        const animal = createAnimalMesh(q.animal);
        const offset = (i - (q.count - 1) / 2) * 0.7;

        if (q.prep === "on") {
          let surfaceY = (q.obj === "table") ? 1.8 : (q.obj === "bed") ? 1.0 : (q.obj === "sofa") ? 0.9 : 1.2;
          animal.position.set(offset, surfaceY, 0);
        } else if (q.prep === "under") {
          animal.position.set(offset, 0.3, 0);
        } else if (q.prep === "in") {
          animal.position.set(offset, 0.8, 0);
        }
        objectGroup.add(animal);
      }
    }

    function createAnimalMesh(type) {
      const group = new THREE.Group();
      let color = 0xF97316; // Cat
      if (type === "dog") color = 0xA16207;
      if (type === "rabbit") color = 0xE2E8F0;
      if (type === "frog") color = 0x22C55E;
      if (type === "bird") color = 0x06B6D4;
      if (type === "kitten") color = 0xEC4899;

      const body = new THREE.Mesh(new THREE.BoxGeometry(0.5, 0.5, 0.5), new THREE.MeshStandardMaterial({color: color}));
      const head = new THREE.Mesh(new THREE.BoxGeometry(0.35, 0.35, 0.35), new THREE.MeshStandardMaterial({color: color}));
      head.position.set(0, 0.4, 0.15);
      group.add(body, head);
      return group;
    }

    // --- INITIALIZE ON LOAD ---
    window.onload = () => {
      init3D();
    };
  </script>
